  5    """
  6    Parses a flat list of tokens into an Abstract Syntax Tree (AST).
  7
  8    The parser uses recursive descent to handle operator precedence and
  9    nested expressions. It consumes tokens produced by the Lexer and
 10    builds a Program node whose children represent every top-level
 11    statement in the source.
 12
 13    Attributes:
 14        tokens (list): The full list of Token objects to parse.
 15        index (int): The current position in the token list.
 16        current_token (Token): The token at the current index.
 17    """
 18
 19    def __init__(self, tokens):
 20        """
 21        Initialize the Parser with a list of tokens.
 22
 23        Args:
 24            tokens (list): A non-empty list of Token objects ending with
 25                an EOF token.
 26
 27        Raises:
 28            Exception: If the token list is empty.
 29        """
 30        self.tokens = tokens
 31        
 32        if self.tokens == []:
 33            raise Exception("bestie... no tokens to parse 💀")
 34
 35        self.index = 0
 36        self.current_token = self.tokens[self.index]
 37
 38    def advance(self):
 39        """
 40        Move to the next token in the list.
 41
 42        Increments the index and updates current_token. If the end of
 43        the token list is reached, current_token is set to None.
 44        """
 45        self.index += 1
 46        if self.index >= len(self.tokens):
 47            self.current_token = None
 48        else:
 49            self.current_token = self.tokens[self.index]
 50
 51    def peek(self):
 52        """
 53        Look at the next token without consuming it.
 54
 55        Returns:
 56            Token | None: The token immediately after the current one,
 57            or None if the current token is the last in the list.
 58        """
 59        if self.index + 1 >= len(self.tokens):
 60            return None
 61        return self.tokens[self.index + 1]
 62
 63    def parse(self):
 64        """
 65        Parse the entire token stream into a Program AST node.
 66
 67        Iterates over all tokens, skipping newlines and delegating each
 68        non-trivial token to parse_statement(), until the EOF token is
 69        reached.
 70
 71        Returns:
 72            Program: The root AST node containing all top-level statements.
 73        """
 74        statements = []
 75
 76        while self.current_token.type != TokenType.EOF:
 77            # skip new lines
 78            if self.current_token.type == TokenType.NEWLINE:
 79                self.advance()
 80            else:
 81                statement = self.parse_statement()
 82                statements.append(statement)
 83        
 84        return Program(statements)
 85    
 86    def parse_statement(self):
 87        """
 88        Parse a single statement based on the current token type.
 89
 90        Dispatches to the appropriate sub-parser depending on what kind
 91        of statement starts at the current token. Identifier tokens are
 92        further disambiguated by peeking at the following token: if it
 93        is an assignment operator the token is treated as the left-hand
 94        side of an assignment, otherwise it is the start of an
 95        expression.
 96
 97        Returns:
 98            ASTNode: One of VarDeclaration, IfStatement, WhileStatement,
 99            FunctionDeclaration, ReturnStatement, or an expression node.
100
101        Raises:
102            Exception: If the current token does not begin a valid statement.
103        """
104        if self.current_token.type in (TokenType.LET, TokenType.CONST):
105            return self.parse_var_declaration()
106        elif self.current_token.type == TokenType.IF:
107            return self.parse_if_statement()
108        elif self.current_token.type == TokenType.WHILE:
109            return self.parse_while_statement()
110        elif self.current_token.type == TokenType.FUNCTION:
111            return self.parse_function_declaration()
112        elif self.current_token.type == TokenType.RETURN:
113            return self.parse_return_statement()
114        elif self.current_token.type in INBUILT_FUNCTIONS:
115            return self.parse_expression()
116        elif self.current_token.type == TokenType.IDENT:
117            if self.peek().type == TokenType.ASSIGN:
118                return self.parse_assignment()
119            else:
120                return self.parse_expression()
121        else:
122            raise Exception(f"my guy.. never seen '{self.current_token.value}' in my life btw")
123
124    def parse_primary(self):
125        """
126        Parse a primary (atomic) expression.
127
128        Handles the smallest indivisible units of an expression:
129        number literals, string literals, boolean literals, identifiers
130        (which may be followed by a call argument list), built-in
131        function calls, and parenthesised sub-expressions.
132
133        Returns:
134            ASTNode: One of NumberLiteral, StringLiteral, BoolLiteral,
135            Identifier, or CallExpression; or the result of a recursive
136            parse_expression() call for parenthesised groups.
137
138        Raises:
139            Exception: If the current token cannot start a primary expression.
140        """
141        if self.current_token.type == TokenType.NUMBER:
142            node = NumberLiteral(self.current_token.value)
143            self.advance()
144            return node
145        elif self.current_token.type == TokenType.STRING:
146            node = StringLiteral(self.current_token.value)
147            self.advance()
148            return node
149        elif self.current_token.type in BOOLEANS:
150            node = BoolLiteral(self.current_token.value)
151            self.advance()
152            return node
153        elif self.current_token.type == TokenType.IDENT:
154            name = self.current_token.value
155            self.advance()
156
157            if self.current_token.type is not None and self.current_token.type == TokenType.LPAREN:
158                args = self.parse_call_args()
159                return CallExpression(name, args)
160            else:
161                return Identifier(name)
162        elif self.current_token.type in INBUILT_FUNCTIONS:
163            name = self.current_token.value  
164            self.advance()
165            
166            if self.current_token.type is not None and self.current_token.type == TokenType.LPAREN:
167                args = self.parse_call_args()
168                return CallExpression(name, args)
169        elif self.current_token.type == TokenType.LPAREN:  
170            self.advance()
171            node = self.parse_expression()
172            self.advance()
173            return node
174        else:
175            raise Exception(f"i dont know how you expect me to parse expression {self.current_token.value}")

177    def parse_var_declaration(self):
178        """
179        Parse a variable declaration statement (let or const).
180
181        Expects the form:  <let|const> <identifier> = <expression>
182
183        Advances past the keyword, reads the variable name, skips the
184        assignment operator, then delegates to parse_expression() for
185        the initialiser value.
186
187        Returns:
188            VarDeclaration: AST node holding the variable name, its
189            initial value expression, and whether it is constant.
190        """
191        constant = self.current_token.type == TokenType.CONST
192        self.advance()
193        name = self.current_token.value
194        self.advance()
195        self.advance()  # skip the '=' token
196        value = self.parse_expression() 
197        return VarDeclaration(name, value, constant)
198    
199    def parse_assignment(self):
200        """
201        Parse a bare assignment to an already-declared variable.
202
203        Expects the form:  <identifier> = <expression>
204
205        This is distinct from parse_var_declaration() in that there is
206        no leading 'let' or 'const' keyword. The resulting node is a
207        non-constant VarDeclaration reusing the existing variable name,
208        which the interpreter is expected to treat as a mutation rather
209        than a fresh binding.
210
211        Returns:
212            VarDeclaration: AST node with constant=False, holding the
213            target variable name and the new value expression.
214        """
215        name = self.current_token.value
216        self.advance()  # skip identifier
217        self.advance()  # skip '='
218        value = self.parse_expression()
219        return VarDeclaration(name, value, False)

221    def parse_call_args(self):
222        """
223        Parse the argument list of a function call.
224
225        Expects the current token to be '(' on entry. Reads comma-
226        separated expressions until the matching ')' is found,
227        then advances past it.
228
229        Returns:
230            list[ASTNode]: A (possibly empty) list of argument nodes,
231            each produced by parse_expression().
232        """
233        self.advance()  # consume '('
234        args = []
235        while self.current_token.type is not None and self.current_token.type != TokenType.RPAREN:
236            if self.current_token.type == TokenType.COMMA:
237                self.advance()
238            args.append(self.parse_expression())
239        self.advance()  # consume ')'
240        return args
241
242    def parse_expression(self):
243        """
244        Parse a binary expression with left-to-right associativity.
245
246        Starts by parsing a primary expression as the left-hand operand,
247        then repeatedly consumes binary operators and right-hand primaries
248        to build a left-associative BinaryExpression tree.
249
250        Note: All operators are treated with equal precedence. If your
251        language requires precedence levels (e.g. * before +), this
252        method will need to be split into separate precedence climbing
253        or Pratt-parsing layers.
254
255        Returns:
256            ASTNode: A single primary node if no binary operator follows,
257            or a (possibly nested) BinaryExpression node otherwise.
258        """
259        left = self.parse_primary()
260
261        while self.current_token is not None and self.current_token.type in BINARY_OPERATORS:
262            operator = self.current_token.value
263            self.advance()
264            right = self.parse_primary()
265            left = BinaryExpression(left, operator, right)
266        
267        return left

269    def parse_block(self):
270        """
271        Parse a brace-delimited block of statements.
272
273        Expects the opening '{' to have already been consumed by the
274        caller. Reads statements until the closing '}' is reached,
275        skipping any newlines in between, then consumes the '}'.
276
277        Returns:
278            list[ASTNode]: The statements found inside the block,
279            in source order.
280        """
281        statements = []
282        while self.current_token.type != TokenType.RBRACE:
283            if self.current_token.type == TokenType.NEWLINE:
284                self.advance()
285            else:
286                statements.append(self.parse_statement())
287        self.advance()  # consume '}'
288        return statements

290    def parse_if_statement(self):
291        """
292        Parse an if / else-if / else statement.
293
294        Handles the full conditional chain: an initial 'sus' branch,
295        zero or more 'mid' (else-if) branches parsed recursively, and
296        an optional 'tho' (else) branch.
297
298        Grammar (informal):
299            sus ( <condition> ) { <body> }
300            [ mid ( <condition> ) { <body> } ]*
301            [ tho { <body> } ]
302
303        Returns:
304            IfStatement: AST node containing the condition, the if-body,
305            and optionally a chained IfStatement (for else-if) or a
306            plain statement list (for else) as else_body.
307        """
308        self.advance()  # skip 'sus'
309        self.advance()  # skip '('
310        condition = self.parse_expression()
311        self.advance()  # skip ')'
312        self.advance()  # skip '{'
313
314        body = self.parse_block()
315        else_body = None
316
317        if self.current_token.type == TokenType.ELSE_IF:
318            else_body = self.parse_if_statement()
319        elif self.current_token.type == TokenType.ELSE:
320            self.advance()  # skip 'tho'
321            self.advance()  # skip '{'
322            else_body = self.parse_block()
323        
324        return IfStatement(condition, body, else_body)
325    
326    def parse_while_statement(self):
327        """
328        Parse a while loop statement.
329
330        Expects the form:  grind (<condition>) { <body> }
331
332        Consumes the 'grind' keyword, the parenthesised condition, and
333        the braced body block.
334
335        Returns:
336            WhileStatement: AST node containing the condition expression
337            and the list of body statements.
338        """
339        self.advance()  # skip 'grind'
340        self.advance()  # skip '('
341        condition = self.parse_expression()
342        self.advance()  # skip ')'
343        self.advance()  # skip '{'
344
345        body = self.parse_block()
346        return WhileStatement(condition, body)
347    
348    def parse_return_statement(self):
349        """
350        Parse a return statement.
351
352        Expects the form:  yeet <expression>
353
354        Consumes the 'yeet' keyword then parses the expression whose
355        value will be returned from the enclosing function. There is no
356        support for bare returns (i.e. 'yeet' with no value); one
357        expression is always required.
358
359        Returns:
360            ReturnStatement: AST node wrapping the return value expression.
361        """
362        self.advance()  # skip 'yeet'
363        return ReturnStatement(self.parse_expression())
364    
365    def parse_function_declaration(self):
366        """
367        Parse a function declaration.
368
369        Expects the form:  cook <name>(<params>) { <body> }
370
371        Consumes the 'cook' keyword, the function name, the
372        comma-separated parameter list (identifiers only, no defaults or
373        type annotations), and the braced body block.
374
375        Returns:
376            FunctionDeclaration: AST node containing the function name,
377            its parameter list as plain strings, and its body statements.
378        """
379        self.advance()  # skip 'cook'
380        name = self.current_token.value
381        self.advance()  # skip function name
382        self.advance()  # skip '('
383        
384        params = []
385        while self.current_token.type != TokenType.RPAREN:
386            if self.current_token.type == TokenType.COMMA:
387                self.advance()
388            params.append(self.current_token.value)
389            self.advance()
390        
391        self.advance()  # skip ')'
392        self.advance()  # skip '{'
393        body = self.parse_block()
394
395        return FunctionDeclaration(name, params, body)

65        """
66        Parse the entire token stream into a Program AST node.
67
68        Iterates over all tokens, skipping newlines and delegating each
69        non-trivial token to parse_statement(), until the EOF token is

71
72        Returns:

103            Exception: If the current token does not begin a valid statement.

109        elif self.current_token.type == TokenType.WHILE:
110            return self.parse_while_statement()
111        elif self.current_token.type == TokenType.FUNCTION:
112            return self.parse_function_declaration()
